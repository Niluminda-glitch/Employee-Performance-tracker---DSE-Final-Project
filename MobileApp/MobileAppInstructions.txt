Database Interaction with Dashboard Features
1. User Management
Database Tables:
users: Stores user information including id, email, password, full_name, role, status, job_title, department_id, supervisor_id, avatar_url, etc.
departments: Stores department information including id, name, description.
Database Interactions:
Employee Dashboard:
Profile Viewing:
GET /api/users/:userId - Retrieves user data from the users table, including related department and supervisor information.
Profile Editing:
PUT /api/users/:userId - Updates specific fields in the users table (full_name, email, job_title, avatar_url).
Supervisor Dashboard:
Team Viewing:
GET /api/supervisor/team - Queries the users table with a WHERE clause on supervisor_id to find all team members.
Includes JOIN with departments table to get department information.
Employee Profile Viewing:
GET /api/users/:userId - Retrieves detailed user information for a specific team member.
Manager Dashboard:
All Users Viewing:
GET /api/manager/users - Retrieves all users from the users table with optional filtering by role.
Includes JOINs with departments and supervisor relationships.
User Creation:
POST /api/manager/users - Inserts a new record into the users table with provided details.
User Editing:
PUT /api/manager/users/:userId - Updates user records in the users table.
User Deactivation/Reactivation:
PUT /api/manager/users/:userId/deactivate - Updates the status field in the users table to 'inactive'.
PUT /api/manager/users/:userId/reactivate - Updates the status field in the users table to 'active'.
2. Task Management
Database Tables:
tasks: Stores task information including id, title, description, assignee_id, supervisor_id, status, priority, due_date, completed_at.
operations: Stores operation details including quality_checkpoints, smv_target, sop_url.
Database Interactions:
Employee Dashboard:
View Assigned Tasks:
GET /api/tasks/my - Queries the tasks table with a WHERE clause on assignee_id matching the current user's id.
Includes JOIN with users table to get supervisor information.
Update Task Status:
PUT /api/tasks/:taskId/status - Updates the status field in the tasks table.
When status is set to 'completed', also updates the completed_at timestamp.
Active Task Assistant:
Retrieves task details including related operation information with quality checkpoints.
Supervisor Dashboard:
View Team Tasks:
GET /api/tasks/team - Complex query that finds tasks where:
assignee_id is in the list of team member IDs (users with supervisor_id matching the current user)
OR supervisor_id matches the current user's id
Includes JOINs with users table to get assignee and supervisor information.
Assign New Tasks:
POST /api/tasks - Inserts a new record into the tasks table with the supervisor_id set to the current user's id.
View User Tasks:
GET /api/tasks/user/:userId - Queries tasks for a specific team member.
Manager Dashboard:
Task Overview:
Similar to supervisor but with broader access to all tasks.
Task Assignment:
Same endpoint as supervisor but with manager role authorization.
3. Request System (Leave and Overtime)

Key Database Relationships
User-Department Relationship:
One-to-Many: A department has many users, but a user belongs to one department.
Foreign Key: department_id in users table references id in departments table.
User-Supervisor Relationship:
Self-referential One-to-Many: A supervisor has many subordinates, but a user has one supervisor.
Foreign Key: supervisor_id in users table references id in the same users table.
User-Task Relationships:
One-to-Many (as assignee): A user has many assigned tasks, but a task has one assignee.
One-to-Many (as supervisor): A user has many created tasks, but a task has one supervisor.
Foreign Keys: assignee_id and supervisor_id in tasks table reference id in users table.
User-Skill Relationship:
Many-to-Many: A user has many skills, and a skill belongs to many users.
Junction Table: user_skills with foreign keys to both users and skills tables.
User-Request Relationship:
One-to-Many: A user has many requests, but a request belongs to one user.
Foreign Key: user_id in requests table references id in users table.
User-Notification Relationship:
One-to-Many: A user has many notifications, but a notification belongs to one user.
Foreign Key: user_id in notifications table references id in users table.

-- Create ENUM Types
CREATE TYPE public.enum_notifications_type AS ENUM (
    'system', 'task_assigned', 'task_completed', 'deadline_reminder',
    'mentorship_request', 'request_approved', 'request_rejected', 'announcement'
);
CREATE TYPE public.enum_requests_status AS ENUM ('pending', 'approved', 'rejected');
CREATE TYPE public.enum_requests_type AS ENUM ('leave', 'overtime');
CREATE TYPE public.enum_tasks_priority AS ENUM ('low', 'medium', 'high');
CREATE TYPE public.enum_tasks_status AS ENUM ('pending', 'in-progress', 'completed');
CREATE TYPE public.enum_user_skills_proficiency_level AS ENUM ('beginner', 'intermediate', 'advanced', 'expert');
CREATE TYPE public.enum_users_role AS ENUM ('employee', 'supervisor', 'manager');
CREATE TYPE public.enum_users_status AS ENUM ('active', 'inactive', 'first_login');
CREATE TYPE public.skill_proficiency AS ENUM ('learning', 'proficient', 'expert');


-- Create Functions
CREATE OR REPLACE FUNCTION public.trigger_set_timestamp() RETURNS trigger
    LANGUAGE plpgsql
    AS $$
BEGIN
  NEW.updated_at = NOW();
  RETURN NEW;
END;
$$;

CREATE OR REPLACE FUNCTION public.update_updated_at_column() RETURNS trigger
    LANGUAGE plpgsql
    AS $$
BEGIN
   IF row(NEW.*) IS DISTINCT FROM row(OLD.*) THEN
      NEW.updated_at = timezone('utc', now());
   END IF;
   RETURN NEW;
END;
$$;


-- Create Tables (Schema Definition - MODIFIED DEFAULT UUID)

CREATE TABLE public.departments (
    id uuid DEFAULT uuid_generate_v4() NOT NULL PRIMARY KEY, -- Corrected default
    name text NOT NULL UNIQUE,
    created_at timestamp with time zone DEFAULT timezone('utc'::text, now()) NOT NULL,
    description text,
    updated_at timestamp with time zone DEFAULT CURRENT_TIMESTAMP
);
COMMENT ON TABLE public.departments IS 'Stores department information';

CREATE TABLE public.users (
    id uuid NOT NULL PRIMARY KEY, -- Users often have IDs from Auth, kept as is
    email text NOT NULL UNIQUE,
    full_name text,
    role public.enum_users_role DEFAULT 'employee'::public.enum_users_role NOT NULL,
    status public.enum_users_status DEFAULT 'first_login'::public.enum_users_status NOT NULL,
    department_id uuid REFERENCES public.departments(id) ON DELETE SET NULL,
    supervisor_id uuid REFERENCES public.users(id) ON DELETE SET NULL,
    avatar_url text,
    password text NOT NULL,
    created_at timestamp with time zone DEFAULT timezone('utc'::text, now()) NOT NULL,
    updated_at timestamp with time zone DEFAULT timezone('utc'::text, now()) NOT NULL,
    job_title text,
    google_id text UNIQUE,
    "createdAt" timestamp with time zone DEFAULT now() NOT NULL -- Consider renaming/removing if redundant
);
COMMENT ON TABLE public.users IS 'Stores public user profile information (and password hash for custom auth)';
COMMENT ON COLUMN public.users.password IS 'Stores the hashed password for custom authentication';
COMMENT ON COLUMN public.users.job_title IS 'Stores the job title of the user';
COMMENT ON COLUMN public.users.google_id IS 'Stores the unique ID from Google Sign-In';

CREATE TABLE public.skills (
    id uuid DEFAULT uuid_generate_v4() NOT NULL PRIMARY KEY, -- Corrected default
    name text NOT NULL UNIQUE,
    description text,
    created_at timestamp with time zone DEFAULT timezone('utc'::text, now()) NOT NULL,
    "createdAt" timestamp with time zone DEFAULT now() NOT NULL,
    "updatedAt" timestamp with time zone DEFAULT now() NOT NULL
);
COMMENT ON TABLE public.skills IS 'Stores available skills in the organization';

CREATE TABLE public.user_skills (
    user_id uuid NOT NULL REFERENCES public.users(id) ON DELETE CASCADE,
    skill_id uuid NOT NULL REFERENCES public.skills(id) ON DELETE CASCADE,
    proficiency_level public.skill_proficiency DEFAULT 'learning'::public.skill_proficiency NOT NULL,
    willing_to_mentor boolean DEFAULT false NOT NULL,
    is_certified boolean DEFAULT false NOT NULL,
    last_certified_date date,
    is_training boolean DEFAULT false NOT NULL,
    created_at timestamp with time zone DEFAULT timezone('utc'::text, now()) NOT NULL,
    PRIMARY KEY (user_id, skill_id)
);
COMMENT ON TABLE public.user_skills IS 'Associates users with skills and their proficiency/mentorship status';

CREATE TABLE public.tasks (
    id uuid DEFAULT uuid_generate_v4() NOT NULL PRIMARY KEY, -- Corrected default
    title text NOT NULL,
    description text,
    assignee_id uuid REFERENCES public.users(id) ON DELETE SET NULL,
    supervisor_id uuid NOT NULL REFERENCES public.users(id) ON DELETE CASCADE,
    status public.enum_tasks_status DEFAULT 'pending'::public.enum_tasks_status NOT NULL,
    priority public.enum_tasks_priority DEFAULT 'medium'::public.enum_tasks_priority NOT NULL,
    due_date date,
    created_at timestamp with time zone DEFAULT timezone('utc'::text, now()) NOT NULL,
    updated_at timestamp with time zone DEFAULT timezone('utc'::text, now()) NOT NULL,
    completed_at timestamp with time zone
);
COMMENT ON TABLE public.tasks IS 'Stores task information';
COMMENT ON COLUMN public.tasks.completed_at IS 'Timestamp when the task status was set to completed';

CREATE TABLE public.notifications (
    id uuid DEFAULT uuid_generate_v4() NOT NULL PRIMARY KEY, -- Corrected default
    user_id uuid NOT NULL REFERENCES public.users(id) ON DELETE CASCADE,
    title text NOT NULL,
    message text NOT NULL,
    type public.enum_notifications_type NOT NULL,
    related_task_id uuid REFERENCES public.tasks(id) ON DELETE SET NULL,
    related_user_id uuid REFERENCES public.users(id) ON DELETE SET NULL,
    is_read boolean DEFAULT false NOT NULL,
    created_at timestamp with time zone DEFAULT timezone('utc'::text, now()) NOT NULL
);
COMMENT ON TABLE public.notifications IS 'Stores user notifications';

CREATE TABLE public.mentorship_requests (
    id uuid DEFAULT uuid_generate_v4() NOT NULL PRIMARY KEY, -- Corrected default
    requester_id uuid NOT NULL REFERENCES public.users(id) ON DELETE CASCADE,
    mentor_id uuid NOT NULL REFERENCES public.users(id) ON DELETE CASCADE,
    skill_id uuid NOT NULL REFERENCES public.skills(id) ON DELETE CASCADE,
    status public.enum_requests_status DEFAULT 'pending'::public.enum_requests_status NOT NULL, -- Reusing requests_status enum
    message text,
    created_at timestamp with time zone DEFAULT timezone('utc'::text, now()) NOT NULL,
    updated_at timestamp with time zone DEFAULT timezone('utc'::text, now()) NOT NULL
);
COMMENT ON TABLE public.mentorship_requests IS 'Tracks mentorship requests between users for specific skills';

CREATE TABLE public.operations (
    id uuid DEFAULT uuid_generate_v4() NOT NULL PRIMARY KEY, -- Corrected default
    name text NOT NULL,
    number text UNIQUE,
    quality_checkpoints jsonb DEFAULT '[]'::jsonb,
    smv_target numeric(5,2),
    sop_url text,
    created_at timestamp with time zone DEFAULT timezone('utc'::text, now()) NOT NULL,
    updated_at timestamp with time zone DEFAULT timezone('utc'::text, now()) NOT NULL
);
COMMENT ON TABLE public.operations IS 'Stores details about specific work operations or tasks';
COMMENT ON COLUMN public.operations.quality_checkpoints IS 'JSON array of objects like [{"icon":"stitch", "text":"Check tension"}, ...]';
COMMENT ON COLUMN public.operations.smv_target IS 'Standard Minute Value target for the operation';

CREATE TABLE public.requests (
    id uuid DEFAULT uuid_generate_v4() NOT NULL PRIMARY KEY, -- Corrected default
    user_id uuid NOT NULL REFERENCES public.users(id) ON DELETE CASCADE,
    type public.enum_requests_type NOT NULL,
    start_date date NOT NULL,
    end_date date NOT NULL,
    hours integer,
    reason text NOT NULL,
    status public.enum_requests_status DEFAULT 'pending'::public.enum_requests_status NOT NULL,
    approver_id uuid REFERENCES public.users(id) ON DELETE SET NULL,
    created_at timestamp with time zone DEFAULT CURRENT_TIMESTAMP NOT NULL,
    updated_at timestamp with time zone DEFAULT CURRENT_TIMESTAMP NOT NULL
);
